cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(ism_cpu C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(ism_cpu src/main.c src/usb_descriptors.c)
target_include_directories(ism_cpu PRIVATE ${CMAKE_CURRENT_LIST_DIR}/src)
target_link_libraries(ism_cpu pico_stdlib tinyusb_device tinyusb_board hardware_gpio hardware_spi hardware_i2c hardware_uart)
pico_enable_stdio_usb(ism_cpu 0)
pico_enable_stdio_uart(ism_cpu 0)
pico_add_extra_outputs(ism_cpu)
target_compile_definitions(ism_cpu PRIVATE CFG_TUSB_MCU=OPT_MCU_RP2040)
