cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(ism_thumbdrives C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(ism_thumbdrives src/main.c src/usb_descriptors.c)
target_include_directories(ism_thumbdrives PRIVATE ${CMAKE_CURRENT_LIST_DIR}/src)
target_link_libraries(ism_thumbdrives pico_stdlib tinyusb_device tinyusb_board hardware_gpio hardware_spi)
pico_enable_stdio_usb(ism_thumbdrives 0)
pico_enable_stdio_uart(ism_thumbdrives 0)
pico_add_extra_outputs(ism_thumbdrives)
target_compile_definitions(ism_thumbdrives PRIVATE CFG_TUSB_MCU=OPT_MCU_RP2040)
