cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(ism_sata C CXX ASM)
pico_sdk_init()

add_executable(ism_sata src/main.c src/usb_descriptors.c)
target_include_directories(ism_sata PRIVATE ${CMAKE_CURRENT_LIST_DIR}/src)
target_link_libraries(ism_sata pico_stdlib tinyusb_device tinyusb_board hardware_gpio)
pico_enable_stdio_usb(ism_sata 0)
pico_enable_stdio_uart(ism_sata 0)
pico_add_extra_outputs(ism_sata)
target_compile_definitions(ism_sata PRIVATE CFG_TUSB_MCU=OPT_MCU_RP2040)
