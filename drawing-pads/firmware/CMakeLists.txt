cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(ism_drawing_pad C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(ism_drawing_pad src/main.c src/usb_descriptors.c)
target_include_directories(ism_drawing_pad PRIVATE ${CMAKE_CURRENT_LIST_DIR}/src)
target_link_libraries(ism_drawing_pad pico_stdlib tinyusb_device tinyusb_board hardware_gpio hardware_adc)
pico_enable_stdio_usb(ism_drawing_pad 0)
pico_enable_stdio_uart(ism_drawing_pad 0)
pico_add_extra_outputs(ism_drawing_pad)
target_compile_definitions(ism_drawing_pad PRIVATE CFG_TUSB_MCU=OPT_MCU_RP2040)
