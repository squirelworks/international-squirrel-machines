cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(ism_typewriter C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(ism_typewriter src/main.c src/usb_descriptors.c src/matrix.c)
target_include_directories(ism_typewriter PRIVATE ${CMAKE_CURRENT_LIST_DIR}/src)
target_link_libraries(ism_typewriter pico_stdlib tinyusb_device tinyusb_board hardware_gpio)
pico_enable_stdio_usb(ism_typewriter 0)
pico_enable_stdio_uart(ism_typewriter 0)
pico_add_extra_outputs(ism_typewriter)
target_compile_definitions(ism_typewriter PRIVATE CFG_TUSB_MCU=OPT_MCU_RP2040)
